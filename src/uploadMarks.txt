const PROGRESS_BAR_LEN = 20;
const MANAGED_FILENAME_PREFIX = 'exam_scan_';

const fs = require('node:fs/promises');
const OPTIONS = {
  headers: {
    'Authorization': `Bearer ${API_KEY}`,
    'Content-Type': 'application/json',
    'Accept': 'application/json+canvas-string-ids'
  }
};

async function checkedFetch(url, options) {
  const response = await fetch(url, options);
  if (!response.ok) {
    throw new Error(`Received ${response.status}: ${response.statusText} from ${url}\n${await response.text()}`);
  }
  return response;
}

function normaliseComment(value) {
  return `${value ?? ''}`
    .replace(/<br\s*\/?\s*>/gi, '\n')
    .replace(/<\/p>/gi, '\n')
    .replace(/<[^>]+>/g, '')
    .replaceAll('&amp;', '&')
    .replaceAll('&lt;', '<')
    .replaceAll('&gt;', '>')
    .replaceAll('&quot;', '"')
    .replaceAll('&#39;', "'")
    .trim();
}

(async () => {
  const grade = async (userId, score, hash, attachment) => {
    const filename = `${MANAGED_FILENAME_PREFIX}${hash}.pdf`;
    const submissionUrl = `https://canvas.qut.edu.au/api/v1/courses/${COURSE_ID}/assignments/${ASSIGNMENT_ID}/submissions/${userId}`;
    const response = await checkedFetch(`${submissionUrl}?include[]=submission_html_comments`, { method: 'GET', ...OPTIONS });
    const submission = await response.json();
    const managedComments = (submission.submission_html_comments ?? []).filter(comment =>
      (comment.attachments ?? []).some(item => item.filename?.startsWith(MANAGED_FILENAME_PREFIX))
    );
    const expectedComment = INCLUDE_COMMENT ? COMMENT : '';
    let canReuse = false;

    for (const comment of managedComments) {
      const existingFilename = comment.attachments?.[0]?.filename;
      const reusable = ATTACH_PDFS
        && `${submission.grade}` === `${score}`
        && existingFilename === filename
        && normaliseComment(comment.comment) === normaliseComment(expectedComment);
      if (reusable) {
        canReuse = true;
        continue;
      }

      for (const item of comment.attachments ?? []) {
        await checkedFetch(`https://canvas.qut.edu.au/api/v1/files/${item.id}`, { method: 'DELETE', ...OPTIONS });
      }
      await checkedFetch(`${submissionUrl}/comments/${comment.id}`, { method: 'DELETE', ...OPTIONS });
    }

    if (canReuse) return;

    if (!ATTACH_PDFS) {
      await checkedFetch(submissionUrl, {
        method: 'PUT',
        body: JSON.stringify({ submission: { posted_grade: Number.parseFloat(score) } }),
        ...OPTIONS
      });
      return;
    }

    const fileBlob = new Blob([await fs.readFile(attachment)]);
    const uploadRequestUrl = `${submissionUrl}/comments/files`;
    const uploadRequest = await checkedFetch(uploadRequestUrl, {
      method: 'POST', body: JSON.stringify({ name: filename, size: fileBlob.size }), ...OPTIONS
    });
    const upload = await uploadRequest.json();
    const formData = new FormData();
    for (const [key, value] of Object.entries(upload.upload_params)) formData.append(key, value);
    formData.append('file', fileBlob);
    const uploadedFileResponse = await checkedFetch(upload.upload_url, { method: 'POST', body: formData });
    const uploadedFile = await uploadedFileResponse.json();

    const comment = { file_ids: [uploadedFile.id], attempt: submission.attempt };
    if (INCLUDE_COMMENT) comment.text_comment = COMMENT;
    await checkedFetch(submissionUrl, {
      method: 'PUT',
      body: JSON.stringify({ comment, submission: { posted_grade: Number.parseFloat(score) } }),
      ...OPTIONS
    });
  };

  process.stdout.write(`Uploading grades and exam scans\n[${' '.repeat(PROGRESS_BAR_LEN)}]\r[`);
  let studentsGraded = 0;
  let progress = 0;
  for (const row of GRADES) {
    await grade(...row);
    studentsGraded++;
    const newProgress = Math.floor(studentsGraded * PROGRESS_BAR_LEN / GRADES.length);
    while (newProgress > progress) {
      process.stdout.write('.');
      progress++;
    }
  }
  process.stdout.write(']\n');
})();
