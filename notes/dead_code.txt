  function filter_detected_bubbles(filteredRows) {
    // Filter out unwanted bubbles from rows based on X-coordinate
    let filteredBubbles = [];

    for (const row of filteredRows) {
      const filteredRow = [];
      for (const element of row) {
        const x = parseInt(element[0]);
        if (
          !(1200 < x && x < 1340) &&
          !(610 < x && x < 740) &&
          !(1800 < x && x < 1900)
        ) {
          filteredRow.push(element);
        }
      }
      filteredBubbles.push(filteredRow);
    }

    // Remove any bubbles less than 140 pixels from the left
    filteredBubbles = filteredBubbles.map(row =>
      row.filter(element => element[0] >= 140)
    );

    return filteredBubbles;
  }

  function fill_missing_bubbles(filteredDa, expectedXPositions = null, tolerance = 20) {
    if (expectedXPositions === null) {
      expectedXPositions = [
        192, 288, 382, 474, 572, 784, 880, 974, 1066, 1162,
        1374, 1470, 1564, 1658, 1752, 1966, 2060, 2156, 2248, 2346
      ];
    }

    const finalResult = [];

    for (const row of filteredDa) {
      // sort row by x
      const r = row.slice().sort((a, b) => a[0] - b[0]);
      const filledRow = [];

      if (r.length < expectedXPositions.length) {
        for (const expectedX of expectedXPositions) {
          const match = r.find(bubble => Math.abs(bubble[0] - expectedX) <= tolerance);
          if (match) {
            filledRow.push(match);
          } else {
            const avgY = Math.floor(r.reduce((sum, b) => sum + b[1], 0) / r.length);
            filledRow.push([expectedX, avgY, 22]);
          }
        }
      } else {
        filledRow.push(...r);
      }

      finalResult.push(filledRow);
    }

    return finalResult;
  }

  function filter_valid_groups(finalGroups, minVerticalSpacing = 86) {
    const validGroups = [];
    let prevY = null;

    for (const group of finalGroups) {
      if (!group || group.length === 0) continue; // skip empty groups

      const y = group[0][1]; // Y-coordinate of the first bubble
      if (prevY === null) {
        validGroups.push(group);
        prevY = y;
      } else {
        const diff = Math.abs(y - prevY);
        if (diff >= minVerticalSpacing) {
          validGroups.push(group);
          prevY = y;
        }
      }
    }

    return validGroups;
  }

  function group_by_columns(rows, threshold = 150) {
    const allColumnGroups = [];

    for (let row of rows) {
      // sort by x (first element)
      row = row.slice().sort((a, b) => a[0] - b[0]);

      const columns = [];
      let currentCol = [];

      for (const circle of row) {
        if (currentCol.length === 0) {
          currentCol.push(circle);
        } else if (Math.abs(parseInt(circle[0]) - parseInt(currentCol[currentCol.length - 1][0])) <= threshold) {
          currentCol.push(circle);
        } else {
          columns.push(currentCol);
          currentCol = [circle];
        }
      }

      if (currentCol.length > 0) columns.push(currentCol);
      allColumnGroups.push(columns);
    }

    return allColumnGroups;
  }

    const predictAnswer = async (filled, left_part, tesseract) => {
    const candidates = CheckCrossedBubbles(filled);
    // Don't use tesseract unless we actually have a crossed bubble
    if (left_part !== null && candidates.length > 0) {
      const cropped = left_part.roi(new cv.Rect(0, 5, left_part.cols, left_part.rows - 10));
      const blurred = new cv.Mat();
      cv.GaussianBlur(cropped, blurred, new cv.Size(5, 5), 0);
      const thresh = new cv.Mat();
      cv.threshold(blurred, thresh, 0, 255, cv.THRESH_BINARY + cv.THRESH_OTSU);

      const kernel = cv.Mat.ones(1, 1, cv.CV_8U);
      const clean = new cv.Mat();
      cv.morphologyEx(thresh, clean, cv.MORPH_OPEN, kernel);
      const blob = await matToBlobURL(clean);

      //const originalWarn = console.warn;
      //console.warn = (...args) => {};
      const result = await tesseract.recognize(blob);
      blurred.delete();
      thresh.delete();
      clean.delete();
      kernel.delete();
      //console.warn = originalWarn;
      URL.revokeObjectURL(blob);

      // const symbols = result.data.symbols || [];
      // const topConfidence = symbols.length ? Math.max(...symbols.map(s => s.confidence)) : 0;

      const text = result.data.text.trim();

      if (text && text.length === 1) {
        //console.log("TESSERACT DETECTED", text);
        // Use the tesseract if there are only crossed bubbles
        let notOnlyCrossed = false;
        for (let k in filled) {
          if (filled[k]["filled"] && !filled[k]["crossed"]) {
            notOnlyCrossed = true;
            break;
          }
        }
        if (!notOnlyCrossed)
          return text;
      }
    }

    if (Object.keys(filled).length > 1) {
      // No letter entered, so pick best choice
      let best_choice = null;
      let min_radius = Infinity;
      for (const k in filled) {
        const info = filled[k];
        const r = info["circle"][2];
        if (r < min_radius) {
          min_radius = r;
          best_choice = k;
        }
      }

      if (candidates.length === 0) {
        /*
        Check if filled in bubbles has a larger enough a difference in size
        compared to the best choice (Smallest bubble) to make it a
        crossed out circle. Return the best choice if it is
        Threshold is set to 6 but can be change
        */
        for (const k in filled) {
          if (Math.abs(filled[best_choice]["circle"][2] - filled[k]["circle"][2]) >= 6) {
            return filled[best_choice]["key"];
          }
          return " ";
        }
      } else {
        return filled[best_choice]["key"];
      }
    }
    if (Object.keys(filled).length === 1) {
      const k = Object.keys(filled)[0];
      return filled[k]["key"];
    }
    return " ";
  };

  
  const CheckCrossedBubbles = (filled) => {
    const candidates = [];
    const CROSSED_RADIUS_THRESHOLD = 35;
    for (const k in filled) {
      const current = filled[k];
      if (current["filled"] === true) {
        if (current["circle"][2] > CROSSED_RADIUS_THRESHOLD) {
          filled[k]["crossed"] = true;
          candidates.push(k);
        } else {
          filled[k]["crossed"] = false;
        }
      }
    }
    return candidates;
  };

  
  const detectFilledInBubbles = (unique_circles, grey, answer_key) => {
    const filled = {};
    for (let i = 0; i < unique_circles.length; i++) {
      const c = unique_circles[i];
      const [x, y, radius] = c;
      const padding = 5;
      const top_left_x = Math.max(x - radius - padding, 0);
      const top_left_y = Math.max(y - radius - padding, 0);
      const bottom_right_x = Math.min(x + radius + padding, grey.cols);
      const bottom_right_y = Math.min(y + radius + padding, grey.rows);

      const cropped_grey = grey.roi(new cv.Rect(top_left_x, top_left_y, bottom_right_x - top_left_x, bottom_right_y - top_left_y));
      const binary_image = new cv.Mat();
      cv.threshold(cropped_grey, binary_image, 150, 255, cv.THRESH_BINARY);

      // Morphological closing (fills small holes and connects nearby white regions)
      const kernel = cv.Mat.ones(3, 3, cv.CV_8U);
      const closing = new cv.Mat();
      cv.morphologyEx(binary_image, closing, cv.MORPH_CLOSE, kernel);

      // Count black pixels (value 0)
      const black_pixel_count = closing.rows * closing.cols - cv.countNonZero(closing);
      if (black_pixel_count > 350) {
        filled[i] = {
          "filled": true,
          "circle": c,
          "crossed": false,
          "key": answer_key[i]
        };
      }

      // Cleanup
      kernel.delete();
      closing.delete();
      binary_image.delete();
      cropped_grey.delete();
    }
    return filled;
  };

  const predict_answer_cas = async (filled, txt_rois, tesseract = null, min_crossed_ocr = 1, abcdepredict = null) => {
    let filled_count = 0, crossed_count = 0, empty_count = 0;
    let ocrs = [];
    for (let i = 0; i < txt_rois.length; i++) ocrs.push(null);
    let raw = {
      filled_weights: filled.map(f => f.filled_weight),
      crossed_weights: filled.map(f => f.crossed_weight),
      empty_weights: filled.map(f => f.empty_weight),
      brightnesses: filled.map(f => f.brightness),
      probas: filled.map(f => f.proba),
      ocrs
    };

    for (let i = 0; i < filled.length; i++) {
      if (filled[i].crossed) crossed_count++;
      else if (filled[i].filled) filled_count++;
      else if (filled[i].empty) empty_count++;
    }
    // Make a single non-empty filled
    if (filled_count === 0 && empty_count === 1) {
      for (let i = 0; i < filled.length; i++) {
        if (!filled[i].empty && !filled[i].crossed) {
          filled[i].filled = true;
          filled_count++;
        }
      }
    }

    // If there's one answer, return it
    if (filled_count === 1) {
      for (let i = 0; i < filled.length; i++) {
        if (filled[i].filled && !filled[i].crossed) return [filled[i].key, raw];
      }
      throw new Error("This shouldn't happen");
    }

    // If there's more than one answer, return the one with the highest weight
    if (filled_count > 1) {
      let best_key = null;
      let best_weight = -Infinity;
      for (let i = 0; i < filled.length; i++) {
        if (!filled[i].crossed) {
          if (filled[i].filled_weight > best_weight) {
            best_key = filled[i].key;
            best_weight = filled[i].filled_weight;
          }
        }
      }
      // Settle ties with proba
      //best_key = filled.filter(f => !f.crossed && f.filled_weight === best_weight).sort((a, b) => b.proba.filled - a.proba.filled)[0].key;
      best_key = filled.filter(f => !f.crossed && f.filled_weight === best_weight)[0].key;
      return [best_key, raw];
    }

    // If there is nothing , OCR
    if (filled_count === 0 && crossed_count >= min_crossed_ocr) {

      for (let i = 0; i < txt_rois.length; i++) ocrs[i] = {};

      let answer = null;
      if (abcdepredict) {
        const resultss = await abcdepredict(txt_rois);
        for (let i = 0; i < txt_rois.length; i++) {
          dbgOutputAppend(await matToBlobURL(txt_rois[i]));
          const results = resultss[i];
          const maxscore = Math.max(...results);
          const answerkey = [null, "A", "B", "C", "D", "E"];
          ocrs[i].p = results;
          for (let j = 0; j < results.length; j++) {
            
            if (results[j] === maxscore) {
              if (answer === null) answer = answerkey[j];
              break;
            }
          }
        }
      }

      if (tesseract) {
        for (let i = 0; i < txt_rois.length; i++) {
          const txt_roi = txt_rois[i];
          //cv.GaussianBlur(txt_roi, blurred, new cv.Size(5, 5), 0);
          //cv.threshold(blurred, thresh, 0, 255, cv.THRESH_BINARY + cv.THRESH_OTSU);
          //cv.morphologyEx(thresh, clean, cv.MORPH_OPEN, kernel);

          const blob = await matToBlobURL(txt_roi);
          const result = await tesseract.recognize(blob);
          //dbgOutputAppend(await matToBlobURL(txt_roi));
          URL.revokeObjectURL(blob);
          //console.log("Result", result);
          ocrs[i].tt = result.data.text.trim();
          ocrs[i].tc = result.data.confidence;
          if (result.data.confidence >= 1) {
            const text = result.data.text.trim();

            if (text) {
              //console.log(`Got ${text} from OCR!`);

              answer = text[0];
              break;
            }
          }
        }
      }

      if (answer !== null) return [answer, raw];
    }
    // No answers at all? Return nothing
    if (filled_count === 0 && crossed_count === 0) {
      return [' ', raw];
    }
    // If there's only crossed and no OCR, pick the one with the highest filled_weight
    {
      let best_key = null;
      let best_weight = -Infinity;
      for (let i = 0; i < filled.length; i++) {
        if (filled[i].crossed && filled[i].filled_weight > best_weight) {
          best_key = filled[i].key;
          best_weight = filled[i].filled_weight;
        }
      }
      //best_key = filled.filter(f => f.crossed && f.filled_weight === best_weight).sort((a, b) => b.proba.filled - a.proba.filled)[0].key;
      best_key = filled.filter(f => f.crossed && f.filled_weight === best_weight)[0].key;
      if (best_key !== null) return [best_key, raw];
    }
    // Return nothing
    return [' ', raw];
  };

  const detect_filled_cas = (roi, cascades) => {
    let objs = new cv.RectVector();
    let rejectLevels = new cv.IntVector();
    let levelWeights = new cv.DoubleVector();
    let msize = new cv.Size(60, 60);
    let fill = {
      filled: false,
      crossed: false,
      empty: false,
      filled_weight: -Infinity,
      crossed_weight: -Infinity,
      empty_weight: -Infinity,
      radius: 0,
      brightness: -1
    };

    // Hough circle
    let circlesMat = new cv.Mat();
    cv.HoughCircles(
      roi,
      circlesMat,
      cv.HOUGH_GRADIENT,
      1, // dp
      15, // minDist
      40, // param1 (higher threshold for Canny edge detector)
      25, // param2 (accumulator threshold)
      10, // minRadius
      40  // maxRadius
    );

    // Filter circles based on radius size and y thresholds
    let circles = convertCircles(circlesMat);
    circlesMat.delete();
    let circ_x = 29.5, circ_y = 29.5;
    if (circles.length > 0) {
      // Biggest circle
      circles.sort((a, b) => b[2] - a[2]);
      const circle = circles[0];
      fill.radius = circle[2];
      [circ_x, circ_y] = [circle[0], circle[1]];
    }
    // Get avg brightness
    let mask = cv.Mat.zeros(roi.rows, roi.cols, cv.CV_8UC1);
    const rad = 21.5;
    cv.circle(mask, new cv.Point(circ_x, circ_y), rad - 4.0, new cv.Scalar(255), -1);

    // Compute mean
    let meanScalar = cv.mean(roi, mask);   // returns [mean, 0, 0, 0]
    fill.brightness = meanScalar[0];
    mask.delete();

    // cascades["filled"].detectMultiScale3(roi, objs, rejectLevels, levelWeights, 1.01, 0, 0, msize, msize, true);
    // if (objs.size() > 0) {
    //   fill.filled_weight = levelWeights.get(0);
    //   fill.filled = true;
    // }
    cascades["crossed"].detectMultiScale3(roi, objs, rejectLevels, levelWeights, 1.01, 0, 0, msize, msize, true);
    if (objs.size() > 0) {
      fill.crossed_weight = levelWeights.get(0);
      fill.crossed = true;
    }
    // cascades["empty"].detectMultiScale3(roi, objs, rejectLevels, levelWeights, 1.01, 0, 0, msize, msize, true);
    // if (objs.size() > 0) {
    //   fill.empty_weight = levelWeights.get(0);
    //   fill.empty = true;
    // }

    rejectLevels.delete();
    levelWeights.delete();
    objs.delete();
    return fill;
  };

          } else {
            return { "crossed": 0.0, "empty": 0.9999921149327404, "filled": 7.885067259623725e-06 };

      }

  function predictProba(features) {
    const coefs = [[9e-06, -0.064803], [5e-06, 0.129411], [-1.4e-05, -0.064612]];
    const intercepts = [17.141577, -17.489292, 0.347714];
    const featureNames = ["crossed", "bright"];
    const classNames = ["crossed", "empty", "filled"];

    const scores = [];

    for (let k = 0; k < coefs.length; k++) {
      let s = intercepts[k];
      for (let j = 0; j < featureNames.length; j++) {
        const name = featureNames[j];
        const fv = features[name] === -Infinity ? -999999 : features[name];
        const x = (fv !== undefined) ? fv : 0.0;
        s += coefs[k][j] * x;
      }
      scores.push(s);
    }

    // Softmax for numerical stability
    const maxScore = Math.max(...scores);
    const expScores = scores.map(s => Math.exp(s - maxScore));
    const sumExp = expScores.reduce((a, b) => a + b, 0.0);

    const probs = {};
    for (let k = 0; k < expScores.length; k++) {
      probs[classNames[k]] = expScores[k] / sumExp;
    }

    return probs;
  }

const setResultFields = (setExamResults, i, fields, diffAction = null) => {
  if (typeof fields === 'function') {
    setExamResults(prev => [
      ...prev.slice(0, i),
      {
        ...prev[i], ...(fields(prev[i])), ...(diffAction === null ? false : {

  function matToBlob(mat) {
    const rgba = new cv.Mat();
    if (mat.type() === cv.CV_8UC1)
      cv.cvtColor(mat, rgba, cv.COLOR_GRAY2RGBA);
    else if (mat.type() === cv.CV_8UC3)
      cv.cvtColor(mat, rgba, cv.COLOR_RGB2RGBA);
    else
      mat.copyTo(rgba);

    const canvas = document.createElement('canvas');
    canvas.width = rgba.cols;
    canvas.height = rgba.rows;
    cv.imshow(canvas, rgba);
    rgba.delete();

    return new Promise((resolve, reject) => {
      canvas.toBlob(
        blob => blob ? resolve(blob) : reject(new Error('toBlob failed')),
        'image/png'
      );
    });
  }
  async function matToBlobURL(mat) {
    return URL.createObjectURL(await matToBlob(mat));
  }
  const add_mat_to_zip = async (zip, filename, mat) => {
    await add_blob_to_zip(zip, filename, await matToBlob(mat));
